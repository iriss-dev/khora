cmake_minimum_required(VERSION 3.20)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(khora_pico C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
option(KHORA_ENABLE_OUTPUTS "Allow ARM with physical interlock and button" OFF)
add_executable(khora_pico main.cpp)
target_compile_definitions(khora_pico PRIVATE KHORA_ENABLE_OUTPUTS=$<BOOL:${KHORA_ENABLE_OUTPUTS}>)
target_compile_options(khora_pico PRIVATE -Wall -Wextra -Werror)
target_link_libraries(khora_pico pico_stdlib hardware_pio hardware_dma hardware_clocks hardware_uart hardware_sync)
pico_generate_pio_header(khora_pico ${CMAKE_CURRENT_LIST_DIR}/parallel.pio)
pico_enable_stdio_usb(khora_pico 1)
pico_enable_stdio_uart(khora_pico 0)
pico_add_extra_outputs(khora_pico)
